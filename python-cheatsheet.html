✂️ Срезы списков


lst = [0, 1, 2, 3, 4, 5, 6, 7]

lst[2]     # 2
lst[2:5]   # [2, 3, 4]
lst[:3]    # [0, 1, 2]
lst[3:]    # [3, 4, 5, 6, 7]
lst[::2]   # [0, 2, 4, 6] (шаг 2)

lst[-2:]   # [6, 7] (последние 2)

📐 List comprehension


# Создание списка через цикл
squares = [x**2 for x in range(5)]
# [0, 1, 4, 9, 16]

# С условием
evens = [x for x in range(10) if x % 2 == 0]
# [0, 2, 4, 6, 8]

# Вложенный цикл
matrix = [[x+y for x in range(3)] 
          for y in range(3)]

# С преобразованием
words = ["hello", "world"]
lengths = [len(w) for w in words]
# [5, 5]

🗝️ Методы словарей


d = {"name": "Alice", "age": 25}

d["name"]       # "Alice"
d.get("age")    # 25
d.get("city", "N/A") # N/A (не существует)
d.keys()        # dict_keys(['name', 'age'])
d.values()      # dict_values(['Alice', 25])
d.items()       # Пары (ключ, значение)

d.pop("age")    # Удалить и вернуть
d.clear()       # Удалить всё

🔁 Итерация по словарю


d = {"a": 1, "b": 2, "c": 3}

# По ключам
for key in d:
    print(key)

# По значениям
for value in d.values():
    print(value)

# По парам
for key, value in d.items():
    print(f"{key}: {value}")

# Comprehension для словаря
squares = {x: x**2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

⚙️ Определение функций


# Базовая функция
def greet(name):
    return f"Hello, {name}!"

# Значения по умолчанию
def add(a, b=0):
    return a + b

# Несколько параметров
def info(name, age, city="Unknown"):
    return f"{name}, {age}, {city}"

# *args - произвольное кол-во
def sum_all(*args):
    return sum(args)

# **kwargs - именованные аргументы
def print_info(**kwargs):
    for key, val in kwargs.items():
        print(f"{key}: {val}")

🎯 Lambda и встроенные


# Lambda функции (анонимные)
square = lambda x: x ** 2
square(5)  # 25

# map() - применить функцию
nums = [1, 2, 3, 4, 5]
result = list(map(lambda x: x*2, nums))
# [2, 4, 6, 8, 10]

# filter() - отфильтровать
evens = list(filter(lambda x: x%2==0, nums))
# [2, 4]

# sorted() с ключом
words = ["apple", "pie", "a"]
sorted(words, key=len)
# ["a", "pie", "apple"]

📦 Декораторы


# Простой декоратор
def my_decorator(func):
    def wrapper():
        print("До")
        func()
        print("После")
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

say_hello()
# Выведет: До, Hello!, После

# С параметрами
def repeat(n):
    def decorator(func):
        def wrapper(*args):
            for _ in range(n):
                func(*args)
        return wrapper
    return decorator
