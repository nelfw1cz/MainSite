Создание

lst = [1, 2, 3]

Добавление

lst.insert(i, x)  # По индексу

Удаление

lst.remove(x)     # По значению
lst.pop(i)        # По индексу

lst.clear()       # Очистить

Методы

lst.index(x)      # Индекс элемента
lst.count(x)      # Количество
lst.sort()        # Сортировать
lst.reverse()     # Развернуть
len(lst)          # Длина

List Comprehension

[x**2 for x in range(10)]
[x for x in lst if x > 0]
[a+b for a in l1 for b in l2]

Файлы

Чтение

with open('file.txt', 'r') as f:
    content = f.read()      # Всё
    lines = f.readlines()   # Список строк
    for line in f:          # По строкам
        print(line)

Запись

with open('out.txt', 'w') as f:
    f.write("Hello")
    f.writelines(lines)

Режимы

'r'  # Чтение
'w'  # Запись (перезаписать)
'a'  # Добавление
'b'  # Бинарный
'+'  # Чтение и запись

Исключения

try-except

try:
    result = 10 / 0
except ZeroDivisionError:
    print("Error!")
except (TypeError, ValueError):
    print("Other error")
except Exception as e:
    print(f"Any error: {e}")
finally:
    print("Always")

raise

raise ValueError("Error msg")
raise  # Перевыбросить
